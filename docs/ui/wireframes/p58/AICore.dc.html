3. React Native Implementation Sketch

AI Core Ring — Component Structure
// AICoreRing.tsx — conceptual structure const AICoreRing = ({ state }: { state: 'idle'|'listening'|'speaking'|'processing' }) => { // Animated values (Reanimated) const scale = useSharedValue(1.0); const glowOpacity = useSharedValue(0.12); const glowBlur = useSharedValue(12); const outerRotation = useSharedValue(0); useEffect(() => { const target = motion.aiCore[state] ?? motion.aiCore.idle; scale.value = withSpring(target.scale, motion.spring3); // AI Breath spring glowOpacity.value = withSpring(target.glowOpacity, motion.spring3); // glowBlur via expo-blur BlurView intensity prop // outerRotation: continuous for 'processing', stops for others if (state === 'processing') { outerRotation.value = withRepeat(withTiming(360, { duration: 4000, easing: Easing.linear }), -1); } }, [state]); // Layer rendering order (back → front): // 1. Holographic ground ellipses + light beam // 2. Outer dashed orbit (layer 5) — not animated // 3. Gold gauge ring (layer 4) — rotates when processing // 4. Glow halo (layer 3) — BlurView intensity = glowBlur // 5. Core fill disc (layer 2) — radial gradient // 6. Center disc + lotus icon + labels (layer 1) // 7. Particle sparkles (Lottie or manual SVG animation) } // Diameter recommendation for 390dp screen: const RING_DIAMETER = 260; // 66% of screen width const GAUGE_RING_STROKE = 16; // dp const GAUGE_RING_RADIUS = RING_DIAMETER / 2 - 20; // 110dp
